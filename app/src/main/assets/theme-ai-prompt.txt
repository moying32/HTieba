使用 Python 标准库打包，不能只改名：
import io, json, zipfile
from pathlib import Path
theme = json.loads(Path("theme.json").read_text(encoding="utf-8"))
raw = json.dumps(theme, ensure_ascii=False, indent=2).encode("utf-8")
assert len(raw) <= 32768
buffer = io.BytesIO()
with zipfile.ZipFile(buffer, "w", zipfile.ZIP_DEFLATED) as archive:
    archive.writestr("theme.json", raw)
result = b"HYDROGEN-THEME/1\n" + buffer.getvalue()
assert len(result) <= 262144
Path(theme["id"] + ".h2theme").write_bytes(result)
